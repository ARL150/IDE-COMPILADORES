// EJERCICIO: Grupo 1 ej.4 — término N de Fibonacci (0, 1, 1, 2, 3, 5, ...) (repetitivo)
// CASO: 1 => 0
// CASO: 2 => 1
// CASO: 7 => 8
// CASO: 10 => 34
main{
  int N, a, b, temp, i;
  cin N;
  a = 0;
  b = 1;
  i = 2;
  while(i < N){
    temp = a + b;
    a = b;
    b = temp;
    i = i + 1;
  };
  if(N == 1) then
    cout a;
  else
    cout b;
  end;
}
