// EJERCICIO: Grupo 8 ej.4 — cuenta los ceros entre los números dados; el fin de datos es -1 (repetitivo)
// CASO: 0 5 0 0 3 -1 => 3
// CASO: -1 => 0
// CASO: 4 2 9 -1 => 0
main{
  int x, ceros;
  ceros = 0;
  cin x;
  while(x != -1){
    if(x == 0) then
      ceros = ceros + 1;
    end;
    cin x;
  };
  cout ceros;
}
