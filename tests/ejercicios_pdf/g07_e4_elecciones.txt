// EJERCICIO: Grupos 7.4, 12.4 — elecciones con 4 candidatos; los votos se teclean desordenados, fin con 0
//   Imprime los votos de cada candidato y su porcentaje respecto al total de votos.
// CASO: 1 3 1 4 2 2 1 4 1 1 1 2 1 3 1 4 0 => Votos candidato 1=8|Votos candidato 2=3|Votos candidato 3=2|Votos candidato 4=3|Porcentaje candidato 1=50.0|Porcentaje candidato 2=18.75|Porcentaje candidato 3=12.5|Porcentaje candidato 4=18.75
// CASO: 0 => Votos candidato 1=0|Votos candidato 2=0|Votos candidato 3=0|Votos candidato 4=0
main{
  int voto, v1, v2, v3, v4, total;
  real p1, p2, p3, p4;
  v1 = 0; v2 = 0; v3 = 0; v4 = 0;
  total = 0;
  cin voto;
  while(voto != 0){
    if(voto == 1) then v1 = v1 + 1; end;
    if(voto == 2) then v2 = v2 + 1; end;
    if(voto == 3) then v3 = v3 + 1; end;
    if(voto == 4) then v4 = v4 + 1; end;
    total = total + 1;
    cin voto;
  };
  cout "Votos candidato 1=" << v1;
  cout "Votos candidato 2=" << v2;
  cout "Votos candidato 3=" << v3;
  cout "Votos candidato 4=" << v4;
  if(total > 0) then
    p1 = v1 * 100.0 / total;
    p2 = v2 * 100.0 / total;
    p3 = v3 * 100.0 / total;
    p4 = v4 * 100.0 / total;
    cout "Porcentaje candidato 1=" << p1;
    cout "Porcentaje candidato 2=" << p2;
    cout "Porcentaje candidato 3=" << p3;
    cout "Porcentaje candidato 4=" << p4;
  end;
}
