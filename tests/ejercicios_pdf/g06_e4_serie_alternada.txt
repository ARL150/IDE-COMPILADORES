// EJERCICIO: Grupo 6 ej.4 — serie 1 - 1/2 + 1/3 - 1/4 + ... ± 1/N (reales, punto fijo de 4 decimales)
// CASO: 1 => 1.0
// CASO: 2 => 0.5
// CASO: 3 => 0.8333
// CASO: 4 => 0.5833
main{
  int N, i;
  real suma;
  cin N;
  suma = 0.0;
  i = 1;
  while(i <= N){
    if(i % 2 == 0) then
      suma = suma - 1.0 / i;
    else
      suma = suma + 1.0 / i;
    end;
    i = i + 1;
  };
  cout suma;
}
