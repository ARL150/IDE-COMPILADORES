// EJERCICIO: Grupo 1 ej.3 — tres enteros en orden creciente=1, 0=No (selectivo)
// SUPUESTO: creciente = no decreciente (A <= B <= C); el PDF no aclara los empates.
// CASO: 1 2 3 => 1
// CASO: 1 3 2 => 0
// CASO: 2 2 5 => 1
// CASO: 3 2 1 => 0
main{
  int A, B, C, resultado;
  cin A;
  cin B;
  cin C;
  if(A <= B) then
    if(B <= C) then
      resultado = 1;
    else
      resultado = 0;
    end;
  else
    resultado = 0;
  end;
  cout resultado;
}
