// EXTRA: while anidados, do-while y do-until (repite mientras la condición sea FALSA)
// CASO: => 25|8|-1|3
main{
  int i, j, s, k, c;
  s = 0;
  i = 1;
  while(i <= 3){
    j = 1;
    while(j <= i){
      s = s + i * j;
      j = j + 1;
    };
    i = i + 1;
  };
  cout s;
  k = 0;
  do {
    k = k + 2;
  } while(k < 7);
  cout k;
  do {
    k = k - 3;
  } until(k < 0);
  cout k;
  c = 0;
  while(c < 100 && !(c == 3)){
    c++;
  };
  cout c;
}
