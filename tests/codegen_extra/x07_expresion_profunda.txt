// EXTRA: expresión con muchos temporales anidados y precedencia mezclada
// CASO: 2 3 10 4 1 5 => 65|11
main{
  int a, b, c, d, e, f, r, s;
  cin a;
  cin b;
  cin c;
  cin d;
  cin e;
  cin f;
  r = ((a + b) * (c - d)) / ((e + 1) * (f - 2)) + a * b * c;
  s = 2 + 3 * 4 - 6 / 2 ^ 1 % 4;
  cout r;
  cout s;
}
